# ROS2 GRCM Sensor Fusion Node
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, PointCloud2
from grcm_msgs.msg import GRCMOutput

class GRCMFusionNode(Node):
    def __init__(self):
        super().__init__('grcm_fusion')
        self.camera_sub = self.create_subscription(
            Image, '/camera/image', self.process, 10)
        self.grcm_pub = self.create_publisher(
            GRCMOutput, '/grcm/output', 10)
        self.client = GRCMClient(endpoint="YOUR_API")

    def process(self, msg):
        emb = self.encode_image(msg)
        result = self.client.forward(image_emb=emb)
        self.grcm_pub.publish(result)

syntax = "proto3";
package grcm;

service GRCMService {
    rpc Forward (ForwardRequest) returns (GRCMResponse);
    rpc Perception (PerceptionRequest) returns (PerceptionResponse);
    rpc StreamInference (stream SensorData) returns (stream GRCMResponse);

}

# Optimus Robot GRCM Integration
from optimus_sdk import Robot, MotorController
from grcm_client import GRCMMiddleware

class OptimusGRCM:
    def __init__(self):
        self.robot = Robot()
        self.grcm = GRCMMiddleware(
            endpoint="YOUR_API_URL",
            auth_token="API_KEY"
        )
    
    async def control_loop(self):
        while True:
            sensors = self.robot.get_sensors()
            result = await self.grcm.forward(
                image_emb=sensors.camera,
                action=sensors.joint_positions
            )
            if result.ethical_halt:
                self.robot.emergency_stop()
            else:
                self.robot.execute(result.action)
